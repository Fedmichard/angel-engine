cmake_minimum_required(VERSION 3.26)

project(AdvancedApp 
    VERSION 1.2.0 
    DESCRIPTION "A modern C++ desktop application"
    LANGUAGES CXX
)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(Threads REQUIRED)

set(SOURCES
    src/main.cpp
    src/utils.cpp
    src/app.cpp
)

add_executable(my_advanced_app ${SOURCES})

target_include_directories(my_advanced_app PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)

target_link_libraries(my_advanced_app PRIVATE Threads::Threads)
